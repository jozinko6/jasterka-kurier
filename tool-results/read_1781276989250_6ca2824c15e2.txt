   256→// ─── Item Configurator ───
   257→function ItemConfigurator({
   258→  item,
   259→  open,
   260→  onOpenChange,
   261→}: {
   262→  item: MenuItem
   263→  open: boolean
   264→  onOpenChange: (open: boolean) => void
   265→}) {
   266→  const cartStore = useCartStore()
   267→  const sizeOptions = item.options.filter(o => o.optionType === 'SIZE' && o.isActive)
   268→  const defaultSize = sizeOptions.find(o => o.isDefault) || sizeOptions[0] || null
   269→  const [selectedSize, setSelectedSize] = useState<MenuItemOption | null>(defaultSize)
   270→  const [selectedExtras, setSelectedExtras] = useState<MenuItemOption[]>([])
   271→  const [selectedRemoves, setSelectedRemoves] = useState<MenuItemOption[]>([])
   272→  const [kitchenNote, setKitchenNote] = useState('')
   273→  const [quantity, setQuantity] = useState(1)
   274→
   275→  const extraOptions = item.options.filter(o => o.optionType === 'EXTRA' && o.isActive)
   276→  const removeOptions = item.options.filter(o => o.optionType === 'REMOVE' && o.isActive)
   277→
   278→  const unitTotal = item.basePrice
   279→    + (selectedSize?.priceDelta || 0)
   280→    + selectedExtras.reduce((sum, e) => sum + e.priceDelta, 0)
   281→
   282→  const totalPrice = unitTotal * quantity
   283→
   284→  const handleAdd = () => {
   285→    cartStore.addItem(item, selectedSize, selectedExtras, selectedRemoves, kitchenNote, quantity)
   286→    toast.success(`${item.name} pridané do košíka`)
   287→    onOpenChange(false)
   288→  }
   289→
   290→  const toggleExtra = (option: MenuItemOption) => {
   291→    setSelectedExtras(prev =>
   292→      prev.find(o => o.id === option.id)
   293→        ? prev.filter(o => o.id !== option.id)
   294→        : [...prev, option]
   295→    )
   296→  }
   297→
   298→  const toggleRemove = (option: MenuItemOption) => {
   299→    setSelectedRemoves(prev =>
   300→      prev.find(o => o.id === option.id)
   301→        ? prev.filter(o => o.id !== option.id)
   302→        : [...prev, option]
   303→    )
   304→  }
   305→
   306→  return (
   307→    <Dialog open={open} onOpenChange={onOpenChange}>
   308→      <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto">
   309→        <DialogHeader>
   310→          <DialogTitle className="text-xl" style={{ color: '#4f7f2a' }}>{item.name}</DialogTitle>
   311→          <DialogDescription>{item.description || 'Nastavte si položku podľa chuti'}</DialogDescription>
   312→        </DialogHeader>
   313→
   314→        <div className="space-y-6 mt-2">
   315→          {/* Size selection */}
   316→          {sizeOptions.length > 0 && (
   317→            <div>
   318→              <Label className="text-base font-semibold mb-3 block">Veľkosť</Label>
   319→              <RadioGroup
   320→                value={selectedSize?.id || ''}
   321→                onValueChange={(val) => {
   322→                  const opt = sizeOptions.find(o => o.id === val)
   323→                  if (opt) setSelectedSize(opt)
   324→                }}
   325→              >
   326→                {sizeOptions.map((opt) => (
   327→                  <div
   328→                    key={opt.id}
   329→                    className="flex items-center space-x-2 rounded-lg border p-3 cursor-pointer hover:bg-muted/50 transition-colors"
   330→                    style={{
   331→                      borderColor: selectedSize?.id === opt.id ? '#4f7f2a' : undefined,
   332→                      backgroundColor: selectedSize?.id === opt.id ? '#f0f7ec' : undefined,
   333→                    }}
   334→                    onClick={() => setSelectedSize(opt)}
   335→                  >
   336→                    <RadioGroupItem value={opt.id} style={{ color: '#4f7f2a' }} />
   337→                    <Label className="flex-1 cursor-pointer font-normal">{opt.name}</Label>
   338→                    {opt.priceDelta > 0 && (
   339→                      <span className="text-sm font-medium" style={{ color: '#4f7f2a' }}>
   340→                        +{formatPrice(opt.priceDelta)}
   341→                      </span>
   342→                    )}
   343→                    {opt.priceDelta === 0 && (
   344→                      <span className="text-sm text-muted-foreground">základná</span>
   345→                    )}
   346→                  </div>
   347→                ))}
   348→              </RadioGroup>
   349→            </div>
   350→          )}
   351→
   352→          {/* Extra ingredients */}
   353→          {extraOptions.length > 0 && (
   354→            <div>
   355→              <Label className="text-base font-semibold mb-3 block">Extra suroviny</Label>
   356→              <div className="space-y-2">
   357→                {extraOptions.map((opt) => (
   358→                  <div
   359→                    key={opt.id}
   360→                    className="flex items-center space-x-2 rounded-lg border p-3 cursor-pointer hover:bg-muted/50 transition-colors"
   361→                    style={{
   362→                      borderColor: selectedExtras.find(o => o.id === opt.id) ? '#4f7f2a' : undefined,
   363→                      backgroundColor: selectedExtras.find(o => o.id === opt.id) ? '#f0f7ec' : undefined,
   364→                    }}
   365→                    onClick={() => toggleExtra(opt)}
   366→                  >
   367→                    <Checkbox
   368→                      checked={!!selectedExtras.find(o => o.id === opt.id)}
   369→                      onCheckedChange={() => toggleExtra(opt)}
   370→                      style={{ color: '#4f7f2a' }}
   371→                    />
   372→                    <Label className="flex-1 cursor-pointer font-normal">{opt.name}</Label>
   373→                    <span className="text-sm font-medium" style={{ color: '#4f7f2a' }}>
   374→                      +{formatPrice(opt.priceDelta)}
   375→                    </span>
   376→                  </div>
   377→                ))}
   378→              </div>
   379→            </div>
   380→          )}
   381→
   382→          {/* Remove ingredients */}
   383→          {removeOptions.length > 0 && (
   384→            <div>
   385→              <Label className="text-base font-semibold mb-3 block">Bez suroviny</Label>
   386→              <div className="space-y-2">
   387→                {removeOptions.map((opt) => (
   388→                  <div
   389→                    key={opt.id}
   390→                    className="flex items-center space-x-2 rounded-lg border p-3 cursor-pointer hover:bg-muted/50 transition-colors"
   391→                    style={{
   392→                      borderColor: selectedRemoves.find(o => o.id === opt.id) ? '#c73325' : undefined,
   393→                      backgroundColor: selectedRemoves.find(o => o.id === opt.id) ? '#fef2f0' : undefined,
   394→                    }}
   395→                    onClick={() => toggleRemove(opt)}
   396→                  >
   397→                    <Checkbox
   398→                      checked={!!selectedRemoves.find(o => o.id === opt.id)}
   399→                      onCheckedChange={() => toggleRemove(opt)}
   400→                    />
   401→                    <Label className="flex-1 cursor-pointer font-normal">{opt.name}</Label>
   402→                  </div>
   403→                ))}
   404→              </div>
   405→            </div>
   406→          )}
   407→
   408→          {/* Kitchen note */}
   409→          <div>
   410→            <Label className="text-base font-semibold mb-2 block">Poznámka pre kuchyňu</Label>
   411→            <Textarea
   412→              placeholder="Špeciálne požiadavky, alergie..."
   413→              value={kitchenNote}
   414→              onChange={(e) => setKitchenNote(e.target.value)}
   415→              className="resize-none"
   416→              rows={2}
   417→            />
   418→          </div>
   419→
   420→          {/* Quantity */}
   421→          <div>
   422→            <Label className="text-base font-semibold mb-2 block">Množstvo</Label>
   423→            <div className="flex items-center gap-3">
   424→              <Button
   425→                variant="outline"
   426→                size="icon"
   427→                onClick={() => setQuantity(Math.max(1, quantity - 1))}
   428→              >
   429→                <Minus className="h-4 w-4" />
   430→              </Button>
   431→              <span className="text-xl font-bold w-12 text-center">{quantity}</span>
   432→              <Button
   433→                variant="outline"
   434→                size="icon"
   435→                onClick={() => setQuantity(quantity + 1)}
   436→              >
   437→                <Plus className="h-4 w-4" />
   438→              </Button>
   439→            </div>
   440→          </div>
   441→
   442→          <Separator />
   443→
   444→          {/* Total and add button */}
   445→          <div className="flex items-center justify-between">
   446→            <div>
   447→              <p className="text-sm text-muted-foreground">Celková cena</p>
   448→              <p className="text-2xl font-bold" style={{ color: '#4f7f2a' }}>
   449→                {formatPrice(totalPrice)}
   450→              </p>
   451→            </div>
   452→            <Button
   453→              size="lg"
   454→              className="gap-2 text-base px-6"
   455→              style={{ backgroundColor: '#4f7f2a', color: 'white' }}
   456→              onMouseEnter={(e) => (e.currentTarget.style.backgroundColor = '#3d6620')}
   457→              onMouseLeave={(e) => (e.currentTarget.style.backgroundColor = '#4f7f2a')}
   458→              onClick={handleAdd}
   459→            >
   460→              <ShoppingCart className="h-5 w-5" />
   461→              Pridať do košíka
   462→            </Button>
   463→          </div>
   464→        </div>
   465→      </DialogContent>
   466→    </Dialog>
   467→  )
   468→}

   470→// ─── Cart Sheet ───
   471→function CartSheet({
   472→  open,
   473→  onOpenChange,
   474→  onCheckout,
   475→}: {
   476→  open: boolean
   477→  onOpenChange: (open: boolean) => void
   478→  onCheckout: () => void
   479→}) {
   480→  const cartStore = useCartStore()
   481→  const { data: zones } = useQuery<DeliveryZone[]>({
   482→    queryKey: ['zones'],
   483→    queryFn: () => fetch('/api/zones').then(r => r.json()),
   484→  })
   485→
   486→  const subtotal = cartStore.getSubtotal()
   487→  const defaultZone = zones?.[0]
   488→  const deliveryFee = defaultZone?.deliveryFee || 0
   489→  const total = subtotal + deliveryFee
   490→
   491→  return (
   492→    <Sheet open={open} onOpenChange={onOpenChange}>
   493→      <SheetContent className="w-full sm:max-w-md flex flex-col">
   494→        <SheetHeader>
   495→          <SheetTitle className="flex items-center gap-2">
   496→            <ShoppingCart className="h-5 w-5" style={{ color: '#4f7f2a' }} />
   497→            Košík ({cartStore.getItemCount()} položiek)
   498→          </SheetTitle>
   499→          <SheetDescription>Vaše vybrané položky</SheetDescription>
   500→        </SheetHeader>
   501→
   502→        <div className="flex-1 overflow-y-auto py-4 custom-scrollbar">
   503→          {cartStore.items.length === 0 ? (
   504→            <div className="text-center py-12 text-muted-foreground">
   505→              <ShoppingCart className="h-12 w-12 mx-auto mb-3 opacity-30" />
   506→              <p>Váš košík je prázdny</p>
   507→              <p className="text-sm mt-1">Pridajte položky z menu</p>
   508→            </div>
   509→          ) : (
   510→            <div className="space-y-3">
   511→              {cartStore.items.map((ci, index) => (
   512→                <Card key={index} className="p-3">
   513→                  <div className="flex justify-between items-start">
   514→                    <div className="flex-1 min-w-0">
   515→                      <p className="font-semibold text-sm">{ci.menuItem.name}</p>
   516→                      <div className="text-xs text-muted-foreground space-y-0.5 mt-1">
   517→                        {ci.selectedSize && <p>Veľkosť: {ci.selectedSize.name}</p>}
   518→                        {ci.selectedExtras.length > 0 && (
   519→                          <p>+ {ci.selectedExtras.map(e => e.name).join(', ')}</p>
   520→                        )}
   521→                        {ci.selectedRemoves.length > 0 && (
   522→                          <p className="text-red-600">- {ci.selectedRemoves.map(r => r.name).join(', ')}</p>
   523→                        )}
   524→                        {ci.kitchenNote && <p>Poznámka: {ci.kitchenNote}</p>}
   525→                      </div>
   526→                    </div>
   527→                    <Button
   528→                      variant="ghost"
   529→                      size="icon"
   530→                      className="h-8 w-8 text-muted-foreground hover:text-red-500"
   531→                      onClick={() => cartStore.removeItem(index)}
   532→                    >
   533→                      <X className="h-4 w-4" />
   534→                    </Button>
   535→                  </div>
   536→                  <div className="flex items-center justify-between mt-2">
   537→                    <div className="flex items-center gap-2">
   538→                      <Button
   539→                        variant="outline"
   540→                        size="icon"

   950→// ─── Order Tracking ───
   951→function OrderTracking({ orderId, onBack }: { orderId: string; onBack: () => void }) {
   952→  const { data: order, isLoading } = useQuery<Order>({
   953→    queryKey: ['order', orderId],
   954→    queryFn: () => fetch(`/api/orders/${orderId}`).then(r => r.json()),
   955→    refetchInterval: 5000,
   956→  })
   957→
   958→  const statusSteps: OrderStatus[] = [
   959→    'NEW', 'ACCEPTED', 'IN_KITCHEN', 'PREPARING', 'READY',
   960→    ...(order?.orderType === 'DELIVERY' ? ['WAITING_FOR_COURIER', 'ASSIGNED_TO_COURIER', 'PICKED_UP', 'ON_THE_WAY'] : []),
   961→    'DELIVERED',
   962→  ]
   963→
   964→  const currentStepIndex = order ? statusSteps.indexOf(order.status) : -1
   965→
   966→  return (
   967→    <div className="flex flex-col h-full">
   968→      <div className="flex items-center gap-3 p-4 border-b">
   969→        <Button variant="ghost" size="icon" onClick={onBack}>
   970→          <ArrowLeft className="h-5 w-5" />
   971→        </Button>

   973→          <h2 className="text-lg font-bold" style={{ color: '#4f7f2a' }}>

   977→        </div>
   978→      </div>
   979→
   980→      <div className="flex-1 overflow-y-auto p-4 space-y-6 custom-scrollbar">
   981→        {isLoading ? (
   982→          <div className="space-y-4">
   983→            <Skeleton className="h-32 rounded-xl" />
   984→            <Skeleton className="h-48 rounded-xl" />
   985→          </div>
   986→        ) : order ? (
   987→          <>
   988→            {/* Status badge */}
   989→            <div className="text-center">
   990→              <Badge className={`text-base px-4 py-2 ${getStatusColor(order.status)}`}>
   991→                {ORDER_STATUS_LABELS[order.status]}
   992→              </Badge>
   993→            </div>

   995→            {/* Status stepper */}
   996→            <div className="space-y-2">
   997→              {statusSteps.map((step, i) => {
   998→                const isCompleted = i <= currentStepIndex
   999→                const isCurrent = i === currentStepIndex
  1000→                const isCancelled = order.status === 'CANCELLED' || order.status === 'REFUNDED'
  1001→
  1002→                return (
  1003→                  <div
  1004→                    key={step}
  1005→                    className={`flex items-center gap-3 p-3 rounded-lg transition-all ${
  1006→                      isCurrent ? 'ring-2' : ''
  1007→                    }`}
  1008→                    style={{
  1009→                      backgroundColor: isCompleted ? (isCancelled && isCurrent ? '#fef2f0' : '#f0f7ec') : '#f9f9f9',
  1010→                      ringColor: isCurrent ? '#4f7f2a' : undefined,
  1011→                    }}
  1012→                  >
  1013→                    <div
  1014→                      className="h-8 w-8 rounded-full flex items-center justify-center flex-shrink-0"
  1015→                      style={{
  1016→                        backgroundColor: isCompleted ? (isCancelled && isCurrent ? '#c73325' : '#4f7f2a') : '#e5e5e5',
  1017→                        color: isCompleted ? 'white' : '#999',
  1018→                      }}
  1019→                    >
  1020→                      {isCompleted ? <Check className="h-4 w-4" /> : <span className="text-xs">{i + 1}</span>}
  1021→                    </div>
  1022→                    <span className={`text-sm font-medium ${isCompleted ? '' : 'text-muted-foreground'}`}>
  1023→                      {ORDER_STATUS_LABELS[step]}
  1024→                    </span>
  1025→                  </div>
  1026→                )
  1027→              })}
  1028→            </div>
  1029→
  1030→            {/* Order details */}
  1031→            <Card className="p-4">
  1032→              <h3 className="font-semibold mb-3">Detail objednávky</h3>
  1033→              <div className="space-y-2 text-sm">
  1034→                <div className="flex justify-between">
  1035→                  <span className="text-muted-foreground">Číslo:</span>
  1036→                  <span className="font-medium">{order.orderNumber}</span>
  1037→                </div>
  1038→                <div className="flex justify-between">
  1039→                  <span className="text-muted-foreground">Typ:</span>
  1040→                  <span>{order.orderType === 'DELIVERY' ? 'Rozvoz' : 'Osobný odber'}</span>
  1041→                </div>
  1042→                <div className="flex justify-between">
  1043→                  <span className="text-muted-foreground">Platba:</span>
  1044→                  <span>
  1045→                    {order.paymentMethod === 'CASH' ? 'Hotovosť'
  1046→                      : order.paymentMethod === 'CARD_ON_DELIVERY' ? 'Karta pri doručení'
  1047→                      : 'Karta pri odbere'}
  1048→                  </span>
  1049→                </div>
